// Firmware Signing
imgtool keygen -t ecdsa-p256 -k mykey.pem
imgtool getpub -k filename.pem -e pem > mykey_pub.pem

python3 ~/ncs/v2.9.1/nrf/scripts/bootloader/hash.py --in nrf5340dk.hex > firmware.sha256

openssl dgst -sha256 -sign mykey.pem firmware.sha256 | python3 ~/ncs/v2.9.1/nrf/scripts/bootloader/asn1parse.py --alg ecdsa --contents signature > zephyr.sig

python3 ~/ncs/v2.9.1/nrf/scripts/bootloader/validation_data.py --input nrf5340dk.hex -o nrf5340dk_signed.hex --output-bin nrf5340dk_signed.bin --offset 0 --signature zephyr.sig --public-key mykey_pub.pem --magic-value "0x281ee6de,0x86518483,0x13502"

// Bootloader Provisioning
python3 ~/ncs/v2.9.1/nrf/scripts/bootloader/provision.py --s0-addr 0x8000 --s1-addr 0x84000 --provision-addr 0xff8100 --public-key-files "mykey_pub.pem" --output provision.hex --max-size 0x280 --num-counter-slots-version 2 --mcuboot-counters-slots 2
